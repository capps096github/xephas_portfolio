# secrets.txt: strings to replace everywhere in your git history
#
# Used by clean-history.sh:  SECRETS_FILE=secrets.txt bash clean-history.sh <repo>
# Guide: https://xephas.me/code/prepare/
#
# One rule per line:   the-real-value==>REPLACEMENT
# The file that held the secret stays; only the string changes, in every commit.
# Lines starting with # are ignored. Delete this file after use: it holds your
# real secrets.
#
# Rotate (cancel and reissue) every key listed here as well. The old value
# still exists in your original branches.

# A plain value:
sk_live_51Hxxxxxxxxxxxxxxxxxxxxxxxx==>YOUR_STRIPE_SECRET_KEY
my-keystore-password==>YOUR_KEYSTORE_PASSWORD

# Real user data found in test files or seed data:
jane.doe@gmail.com==>user@example.com
+256700000000==>+256000000000

# A pattern (Python regex), prefix with regex:
regex:AKIA[0-9A-Z]{16}==>YOUR_AWS_ACCESS_KEY_ID
